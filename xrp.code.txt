def handle_heading_rollover(current_h, previous_h):
    diff = current_h - (previous_h % 360)
    if diff > 180:
        diff -= 360
    elif diff < -180:
        diff += 360
    return previous_h + diff

def set_motors(left, right):
    left_motor.set_effort(left)
    right_motor.set_effort(right * right_motor_handicap)

# ==========================================
# --- PROPORTIONAL TURN CONTROLLER       ---
# ==========================================
def turn_to_heading(target_precision=0.5):
    global integral_error 
    error = ideal_heading - current_heading
    
    if error > 180: error -= 360
    elif error < -180: error += 360
        
    if abs(error) <= target_precision:
        set_motors(0, 0)
        integral_error = 0.0 
        return True 
        
    Kp_turn = 0.015
    dynamic_effort = abs(error) * Kp_turn
    
    if dynamic_effort < 0.38: 
        dynamic_effort = 0.38
    elif dynamic_effort > 0.65: 
        dynamic_effort = 0.65
        
    if error > 0:
        set_motors(-dynamic_effort, dynamic_effort)
    else:
        set_motors(dynamic_effort, -dynamic_effort)
        
    return False 

# ==========================================
# --- FULL PID CONTROLLER ENGINE         ---
# ==========================================
def drive_straight_pid(base_effort=nominal_effort):
    global integral_error, previous_error, last_time
    
    current_time = time.time()
    dt = current_time - last_time
    if dt <= 0.001: dt = 0.001 
    
    error = ideal_heading - current_heading
    
    # 360-Degree Heading Wrap-around Fix
    if error > 180: error -= 360
    elif error < -180: error += 360
    
    if abs(error) > 15:
        spin_effort = 0.35
        if error > 0:
            set_motors(-spin_effort, spin_effort)
        else:
            set_motors(spin_effort, -spin_effort)
        
        integral_error = 0.0
        previous_error = error
        last_time = current_time
        return

    integral_error += error * dt
    derivative = (error - previous_error) / dt
    
    if integral_error > 50: integral_error = 50
    elif integral_error < -50: integral_error = -50
    
    Kp = 0.006  
    Ki = 0.001  
    Kd = 0.001  
    
    correction = (Kp * error) + (Ki * integral_error) + (Kd * derivative)
    
    if correction > 0.08: correction = 0.08
    elif correction < -0.08: correction = -0.08
        
    set_motors(base_effort - correction, base_effort + correction)
    
    previous_error = error
    last_time = current_time

# ==========================================
# --- HYBRID WALL FOLLOWER               ---
# ==========================================
def follow_wall_proportional(target_dist=120.0, side="right", base_effort=nominal_effort):
    turn_away_effort = base_effort * 1.3
    turn_toward_effort = base_effort * 0.8

    if side == "right":
        if sensors["right_prox"] >= 300:
            set_motors(turn_toward_effort, turn_away_effort)
        elif sensors["right_prox"] <= 75:
            set_motors(turn_away_effort, turn_toward_effort)
        else:
            drive_straight_pid(base_effort)
            
    else: 
        if sensors["left_prox"] >= 300:
            set_motors(turn_away_effort, turn_toward_effort)
        elif sensors["left_prox"] <= 75:
            set_motors(turn_toward_effort, turn_away_effort)
        else:
            drive_straight_pid(base_effort)

            state = TURN_AWAY_FROM_R2_LIGHT

    elif state == TURN_AWAY_FROM_R2_LIGHT:
        if turn_to_heading():
            time.sleep(0.15)
            state = DRIVE_FAR_SIDE_R2_RET

    elif state == DRIVE_FAR_SIDE_R2_RET:
        drive_straight_pid()
        if sensors["front"] < STOP_DIST:
            set_motors(0, 0)
            if r2_exit_dir == 1: 
                ideal_heading -= 90
            else: 
                ideal_heading += 90
            state = TURN_SOUTH_R2_RET

    elif state == TURN_SOUTH_R2_RET:
        if turn_to_heading():
            time.sleep(0.15)
            state = DRIVE_TO_R1_DIVIDER_RET

    elif state == DRIVE_TO_R1_DIVIDER_RET:
        drive_straight_pid()
        if sensors["front"] < STOP_DIST:
            set_motors(0, 0)
            if r2_exit_dir == 1: 
                ideal_heading -= 90
            else: 
                ideal_heading += 90
            state = TURN_ALONG_R1_DIVIDER_RET

    elif state == TURN_ALONG_R1_DIVIDER_RET:
        if turn_to_heading():
            time.sleep(0.15)
            if r2_exit_dir == 1: 
                state = FIND_R1_GAP_LEFT_RET 
            else:
                state = FIND_R1_GAP_RET 

    # ==========================================
    # --- UNIFIED R1 RETURN PATH             ---
    # ==========================================
    elif state == FIND_R1_GAP_RET:
        if sensors["right_prox"] <= 15:
            set_motors(0, 0)
            desired_position = left_motor.get_position() + 0.75 
            state = PUSH_R1_GAP_RET
        else:
            follow_wall_proportional(target_dist=120.0, side="right")

    elif state == PUSH_R1_GAP_RET:
        drive_straight_pid()
        if left_motor.get_position() > desired_position:
            set_motors(0, 0)
            ideal_heading -= 90 
            state = TURN_INTO_R1

    elif state == FIND_R1_GAP_LEFT_RET:
        if sensors["left_prox"] <= 15:
            set_motors(0, 0)
            desired_position = left_motor.get_position() + 0.75 
            state = PUSH_R1_GAP_LEFT_RET
        else:
            follow_wall_proportional(target_dist=120.0, side="left")

    elif state == PUSH_R1_GAP_LEFT_RET:
        drive_straight_pid()
        if left_motor.get_position() > desired_position:
            set_motors(0, 0)
            ideal_heading += 90 
            state = TURN_INTO_R1

    elif state == TURN_INTO_R1:
        if turn_to_heading():
            time.sleep(0.15)
            desired_position = left_motor.get_position() + 1.0 
            state = FORWARD_INTO_R1

    elif state == FORWARD_INTO_R1:
        drive_straight_pid()
        if left_motor.get_position() > desired_position:
            set_motors(0, 0)
            # CROSSOVER CHECK
            if R1_GAP_POSITION == START_POSITION: 
                state = APPROACH_START_LIGHT
            else: 
                if START_POSITION == "RIGHT": 
                    ideal_heading += 90 
                else: 
                    ideal_heading -= 90 
                state = TURN_FAR_SIDE_R1

    elif state == TURN_FAR_SIDE_R1:
        if turn_to_heading():
            time.sleep(0.15)
            state = DRIVE_FAR_SIDE_R1

    elif state == DRIVE_FAR_SIDE_R1:
        if sensors["front"] < STOP_DIST:
            set_motors(0, 0)
            if START_POSITION == "RIGHT": 
                ideal_heading -= 90 
            else: 
                ideal_heading += 90 
            state = TURN_SOUTH_R1
        else:
            drive_straight_pid()

    elif state == TURN_SOUTH_R1:
        if turn_to_heading():
            time.sleep(0.15)
            state = APPROACH_START_LIGHT

    # ==========================================
    # --- FINAL BUMP EXECUTION               ---
    # ==========================================
    elif state == APPROACH_START_LIGHT:
        drive_straight_pid()
        if sensors["front"] < LIGHT_STOP_DIST:
            set_motors(0, 0)
            desired_position = left_motor.get_position() + 0.6
            state = BUMP_START_LIGHT

    elif state == BUMP_START_LIGHT:
        drive_straight_pid(0.25)
        if left_motor.get_position() > desired_position:
            set_motors(0, 0)
            state = STOP

    elif state == STOP:
        set_motors(0, 0)

    time.sleep(0.02)
